export async function executeDocumentExtraction(documentText: string) {
  for (const modelName of MODEL_CASCADE) {
    try {
      const model = genAI.getGenerativeModel({
        model: modelName,
        generationConfig: {
          responseMimeType: 'application/json',
          responseSchema: {
            type: SchemaType.ARRAY,
            description: 'List of verified active recall quizzes',
            items: {
              type: SchemaType.OBJECT,
              properties: {
                question: { type: SchemaType.STRING },
                answer: { type: SchemaType.STRING },
                concept: { type: SchemaType.STRING },
                difficulty: { type: SchemaType.STRING }
              },
              required: ['question', 'answer', 'concept']
            }
          }
        }
      });

      const result = await model.generateContent([
        { text: `Extract key active-recall questions from:\n${documentText}` }
      ]);

      return JSON.parse(result.response.text());
    } catch (err) {
      console.warn(`Fallback triggered: ${modelName} failed, cascading to next model.`);
    }
  }
  throw new Error('All AI cascade pipelines exhausted.');
}
